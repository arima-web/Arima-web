const CONFIG = { SPREADSHEET_ID: '1D0cl-8Mby1Un2BN_EjmxtcDioKGjEMhpsee_JNJFR40', SHEET_NAME: 'Resumen', SUPABASE_URL: 'https://mjbsqkchyjfqncrgtfsv.supabase.co', TABLE: 'finance_summary' }; // Convierte importes españoles a números function parseNumber(value) { if (value === null || value === undefined || value === '') return 0; if (typeof value === 'number') return value; let text = String(value).trim(); if (text === '-') return 0; const negative = /^\(.*\)$/.test(text); text = text .replace(/[€%\s()]/g, '') .replace(/\./g, '') .replace(',', '.'); const number = Number(text); if (!Number.isFinite(number)) return 0; return negative ? -number : number; } // Sincroniza la pestaña Resumen con Supabase function syncArimaFinance() { const properties = PropertiesService.getScriptProperties(); const supabaseKey = properties.getProperty('SUPABASE_SERVICE_ROLE_KEY'); if (!supabaseKey) { throw new Error('No se encuentra la clave secreta de Supabase.'); } const spreadsheet = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID); const sheet = spreadsheet.getSheetByName(CONFIG.SHEET_NAME); if (!sheet) { throw new Error('No se encuentra la pestaña Resumen.'); } const data = sheet.getDataRange().getDisplayValues(); // Buscar encabezados const headerIndex = data.findIndex(row => row.some(cell => cell.trim().toLowerCase() === 'proyecto') ); if (headerIndex === -1) { throw new Error('No se encuentra el encabezado Proyecto.'); } const headers = data[headerIndex].map(h => h.trim()); const fields = [ 'Proyecto', 'Cliente / financiador', 'Ingresos', 'Gastos directos', 'Resultado antes de horas', 'Horas Beñat', 'Horas Aitor', 'Horas Amets', 'Horas totales', 'Horas valoradas', 'Resultado tras horas', 'Margen', '€/h (antes de horas)', 'IVA a ingresar', 'Pendiente de cobro', 'Pendiente de pago' ]; const numericFields = fields.slice(2); const projects = []; let totals = null; // Leer proyectos y total general for (let i = headerIndex + 1; i < data.length; i++) { const row = data[i]; const projectIndex = headers.indexOf('Proyecto'); if (projectIndex === -1) continue; const projectName = row[projectIndex].trim(); if (!projectName) continue; const item = {}; fields.forEach(field => { const columnIndex = headers.indexOf(field); const value = columnIndex >= 0 ? row[columnIndex] : ''; item[field] = numericFields.includes(field) ? parseNumber(value) : value; }); if (projectName.toUpperCase() === 'TOTAL ARIMA') { totals = item; continue; } if (projectName.toUpperCase().includes('EQUIPO Y TARIFA')) { continue; } if (projectName.toUpperCase() === 'MIEMBRO') { continue; } projects.push(item); } // Leer tarifas internas del equipo const team = []; const teamStart = data.findIndex(row => row.some(cell => cell.toUpperCase().includes('EQUIPO Y TARIFA') ) ); if (teamStart >= 0) { for (let i = teamStart + 1; i < data.length; i++) { const member = data[i][0].trim(); if (!member) continue; if (['BEÑAT', 'AITOR', 'AMETS'].includes(member.toUpperCase())) { team.push({ member: member, hourlyRate: parseNumber(data[i][1]) }); } } } // Preparar información const payload = { source: 'Cuentas Arima', sheet: 'Resumen', syncedAt: new Date().toISOString(), projects: projects, totals: totals, team: team }; // Enviar a Supabase const url = CONFIG.SUPABASE_URL + '/rest/v1/' + CONFIG.TABLE + '?on_conflict=id'; const response = UrlFetchApp.fetch(url, { method: 'post', contentType: 'application/json', headers: { apikey: supabaseKey, Authorization: 'Bearer ' + supabaseKey, Prefer: 'resolution=merge-duplicates,return=minimal' }, payload: JSON.stringify({ id: 'arima-main', payload: payload, updated_at: new Date().toISOString() }), muteHttpExceptions: true }); const status = response.getResponseCode(); if (status < 200 || status >= 300) { throw new Error( 'Error Supabase ' + status + ': ' + response.getContentText() ); } Logger.log('ECONOMÍA ARIMA SINCRONIZADA CORRECTAMENTE'); Logger.log('Proyectos importados: ' + projects.length); } // Activar actualización automática cada 15 minutos function activarSincronizacionAutomatica() { const triggers = ScriptApp.getProjectTriggers(); triggers.forEach(trigger => { if (trigger.getHandlerFunction() === 'syncArimaFinance') { ScriptApp.deleteTrigger(trigger); } }); ScriptApp.newTrigger('syncArimaFinance') .timeBased() .everyMinutes(15) .create(); // Primera sincronización syncArimaFinance(); Logger.log('Actualización automática activada.'); }
